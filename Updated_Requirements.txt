create table public.resume (
  id bigint generated by default as identity not null,
  title text not null,
  type text not null,
  file_url text not null,
  file_name text null,
  is_primary boolean null default false,
  created_at timestamp with time zone null default now(),
  updated_at timestamp with time zone null default now(),
  constraint resume_pkey primary key (id),
  constraint resume_type_check check (
    (
      type = any (array['pdf'::text, 'word'::text, 'link'::text])
    )
  )
) TABLESPACE pg_default;

create index IF not exists resume_is_primary_idx on public.resume using btree (is_primary) TABLESPACE pg_default;

create index IF not exists resume_created_at_idx on public.resume using btree (created_at desc) TABLESPACE pg_default;
